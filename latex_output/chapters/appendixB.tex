\section{代码组分类}

\subsection{前照明组}

\begin{itemize}[leftmargin=2em]
    \item 远光灯：B, D, HL
    \item 近光灯：C, HL
    \item 自动远光：A, HL
    \item 前雾灯：F, FFS
    \item 前位置灯：P
    \item 昼间行驶灯：DRL
\end{itemize}

\subsection{后照明组}

\begin{itemize}[leftmargin=2em]
    \item 后位置灯：R, ZL
    \item 后雾灯：F2（草案版新增）
    \item 倒车灯：CR, RL
    \item 制动灯：S, S1, S2, S3, S4
    \item 紧急制动信号：E, ESS
\end{itemize}

\subsection{侧照明组}

\begin{itemize}[leftmargin=2em]
    \item 侧标志灯：SL
    \item 侧反射器：SR
    \item 侧转向灯：T (侧)
\end{itemize}

\subsection{信号组}

\begin{itemize}[leftmargin=2em]
    \item 转向灯：T
    \item 危险警告：W
\end{itemize}

\subsection{反射器组}

\begin{itemize}[leftmargin=2em]
    \item 后反射器：LR, RST
    \item 侧反射器：SR
    \item 前反射器：PR（草案版新增）
\end{itemize}

\subsection{装饰组（草案版新增）}

\begin{itemize}[leftmargin=2em]
    \item Logo 投影灯：FL
    \item 迎宾灯：I, WL
\end{itemize}

\subsection{智能照明组（草案版新增）}

\begin{itemize}[leftmargin=2em]
    \item ADB 系统：A, HL
    \item 矩阵式 LED：XML
    \item AFS 系统：AFS
\end{itemize}

\section{代码使用注意事项}

\begin{enumerate}[leftmargin=2em]
    \item 三版本代码体系存在一定差异，互译时需谨慎；
    \item 草案版新增的代码（FL, I, XML 等）在现行版中无对应；
    \item UN R48 的代码体系与中国 GB 4785 的代码体系存在差异；
    \item 在实际使用中，应以法规原文的代码为准。
\end{enumerate}
